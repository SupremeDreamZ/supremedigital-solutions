const safeStorage = {
  get(key, fallback = null) {
    try {
      const value = window.localStorage.getItem(key);
      return value === null ? fallback : value;
    } catch (error) {
      return fallback;
    }
  },
  set(key, value) {
    try {
      window.localStorage.setItem(key, value);
      return true;
    } catch (error) {
      return false;
    }
  }
};
function readStoredArray(key) {
  try {
    const value = JSON.parse(safeStorage.get(key, '[]'));
    return Array.isArray(value) ? value : [];
  } catch (error) {
    return [];
  }
}

let toastTimer = 0;
let audioContext = null;
let masterGain = null;

function haptic(pattern = 12) {
  try {
    if ('vibrate' in navigator) navigator.vibrate(pattern);
  } catch (error) {
    // Haptics are optional. Browsers, in their wisdom, may ignore them.
  }
}

function showToast(message, duration = 2400) {
  const toast = $('#toast');
  if (!toast) return;
  clearTimeout(toastTimer);
  toast.textContent = message;
  toast.hidden = false;
  toastTimer = window.setTimeout(() => {
    toast.hidden = true;
  }, duration);
}

function ensureAudio() {
  if (audioContext) return audioContext;
  const AudioContextClass = window.AudioContext || window.webkitAudioContext;
  if (!AudioContextClass) return null;
  audioContext = new AudioContextClass();
  masterGain = audioContext.createGain();
  masterGain.gain.value = 0.065;
  masterGain.connect(audioContext.destination);
  return audioContext;
}

function tone(frequency = 440, duration = 0.07, type = 'triangle', endFrequency = null, gain = 0.18) {
  if (!soundEnabled) return;
  const context = ensureAudio();
  if (!context || !masterGain) return;
  if (context.state === 'suspended') context.resume();
  const oscillator = context.createOscillator();
  const envelope = context.createGain();
  const now = context.currentTime;
  oscillator.type = type;
  oscillator.frequency.setValueAtTime(frequency, now);
  if (endFrequency) oscillator.frequency.exponentialRampToValueAtTime(Math.max(1, endFrequency), now + duration);
  envelope.gain.setValueAtTime(0.0001, now);
  envelope.gain.exponentialRampToValueAtTime(gain, now + 0.008);
  envelope.gain.exponentialRampToValueAtTime(0.0001, now + duration);
  oscillator.connect(envelope);
  envelope.connect(masterGain);
  oscillator.start(now);
  oscillator.stop(now + duration + 0.02);
}

function noise(duration = 0.05, gain = 0.05, highpass = 800) {
  if (!soundEnabled) return;
  const context = ensureAudio();
  if (!context || !masterGain) return;
  const length = Math.max(1, Math.floor(context.sampleRate * duration));
  const buffer = context.createBuffer(1, length, context.sampleRate);
  const data = buffer.getChannelData(0);
  for (let index = 0; index < length; index += 1) {
    data[index] = (Math.random() * 2 - 1) * (1 - index / length);
  }
  const source = context.createBufferSource();
  const filter = context.createBiquadFilter();
  const envelope = context.createGain();
  source.buffer = buffer;
  filter.type = 'highpass';
  filter.frequency.value = highpass;
  envelope.gain.value = gain;
  source.connect(filter);
  filter.connect(envelope);
  envelope.connect(masterGain);
  source.start();
}

function updateValidationUI() {
  const stamp = $('#validationStamp');
  const count = $('#validationStampCount');
  if (count) count.textContent = `${validation.size}/4`;
  if (stamp) stamp.hidden = validation.size === 0;
  $$('.loadout article').forEach((card) => {
    const unlocked = validation.has(card.dataset.module);
    card.classList.toggle('unlocked', unlocked);
    card.setAttribute('aria-pressed', String(unlocked));
  });
  $$('.token-hotspot').forEach((button) => {
    button.setAttribute('aria-pressed', String(validation.has(button.dataset.token)));
  });
  safeStorage.set('powrhouse-validation', JSON.stringify([...validation]));
}

function scrollPosterTo(pixelY) {
  const frame = $('#posterFrame');
  if (!frame) return;
  const rect = frame.getBoundingClientRect();
  const top = window.scrollY + rect.top;
  const target = top + (pixelY / posterHeight) * rect.height;
  window.scrollTo({ top: target, behavior: 'smooth' });
}

function initAgeGate() {
  const gate = $('#ageGate');
  if (!gate) return;
  const enter = $('#ageEnter');
  const leave = $('#ageLeave');
  const background = [document.querySelector('main'), document.querySelector('footer'), $('#posterMenu'), $('#strainDialog'), $('#toast')].filter(Boolean);
  const setOpen = (open) => {
    gate.hidden = !open;
    document.body.classList.toggle('locked', open);
    background.forEach((element) => { element.inert = open; });
    if (open && enter) {
      enter.focus({ preventScroll: true });
      requestAnimationFrame(() => enter.focus({ preventScroll: true }));
      window.setTimeout(() => enter.focus({ preventScroll: true }), 0);
    }
  };

  const verified = safeStorage.get('powrhouse-21') === 'yes';
  setOpen(!verified);

  gate.addEventListener('keydown', (event) => {
    if (event.key !== 'Tab') return;
    const controls = [enter, leave].filter(Boolean);
    if (!controls.length) return;
    const first = controls[0];
    const last = controls[controls.length - 1];
    if (event.shiftKey && document.activeElement === first) {
      event.preventDefault();
      last.focus();
    } else if (!event.shiftKey && document.activeElement === last) {
      event.preventDefault();
      first.focus();
    }
  });

  enter?.addEventListener('click', () => {
    safeStorage.set('powrhouse-21', 'yes');
    setOpen(false);
    document.body.classList.remove('hotspot-hover-armed');
    soundEnabled = true;
    ensureAudio()?.resume();
    tone(260, 0.12, 'triangle', 520, 0.16);
    requestAnimationFrame(() => $('#poster')?.focus({ preventScroll: true }));
  });

  leave?.addEventListener('click', () => {
    if (window.history.length > 1) window.history.back();
    else window.location.href = 'https://www.powr.house/';
  });
}

function initPosterInteractions() {
  const menu = $('#posterMenu');
  const dialog = $('#strainDialog');
  const menuClose = $('#closeMenu');
  let menuReturnFocus = null;
  const menuBackground = [document.querySelector('main'), document.querySelector('footer'), dialog, $('#toast')].filter(Boolean);
  const openMenu = (trigger) => {
    if (!menu) return;
    menuReturnFocus = trigger || document.activeElement;
    menu.hidden = false;
    menuBackground.forEach((element) => { element.inert = true; });
    requestAnimationFrame(() => menuClose?.focus());
  };
  const closeMenu = () => {
    if (!menu || menu.hidden) return;
    menu.hidden = true;
    menuBackground.forEach((element) => { element.inert = false; });
    if (menuReturnFocus instanceof HTMLElement) menuReturnFocus.focus({ preventScroll: true });
  };
  menu?.addEventListener('keydown', (event) => {
    if (event.key === 'Escape') {
      event.preventDefault();
      closeMenu();
      return;
    }
    if (event.key !== 'Tab') return;
    const controls = $$('button:not([disabled]),a[href]', menu).filter((element) => !element.hidden);
    const first = controls[0];
    const last = controls[controls.length - 1];
    if (event.shiftKey && document.activeElement === first) {
      event.preventDefault();
      last.focus();
    } else if (!event.shiftKey && document.activeElement === last) {
      event.preventDefault();
      first.focus();
    }
  });

  $$('[data-action]').forEach((button) => {
    button.addEventListener('click', () => {
      const action = button.dataset.action;
      if (action === 'cart') {
        showToast('CONCEPT EXPERIENCE ONLY. NO CANNABIS TRANSACTIONS HAPPEN HERE.');
        tone(220, 0.06, 'square', 170, 0.08);
      }
      if (action === 'menu') openMenu(button);
      if (action === 'shop') scrollPosterTo(1120);
      if (action === 'plant') {
        showToast('HERO COLA // LAYERED FLOWER, PURPLE STRUCTURE, ORANGE PISTILS.');
        tone(310, 0.08, 'triangle', 470, 0.1);
      }
      if (action === 'grow') {
        showToast('GROW ROOM // LIVING SOIL, CLEAN INPUTS, SMALL-BATCH ATTENTION.');
        tone(280, 0.08, 'triangle', 380, 0.1);
      }
      if (action === 'retailers') {
        window.open('https://www.powr.house/', '_blank', 'noopener,noreferrer');
      }
    });
  });

  $$('.token-hotspot').forEach((button) => {
    button.addEventListener('click', () => {
      const token = button.dataset.token;
      if (validation.has(token)) {
        validation.delete(token);
        showToast(`${token.toUpperCase()} REMOVED FROM LOADOUT.`);
        tone(300, 0.06, 'triangle', 220, 0.08);
      } else {
        validation.add(token);
        const label = {
          genetics: 'GENETICS // PRECISION SHOT',
          tested: 'TESTED // SPREAD SHOT',
          consistency: 'CONSISTENCY // ROOT SHIELD',
          burn: 'WHITE ASH // SCREEN PURGE'
        }[token];
        showToast(`${label} UNLOCKED.`);
        tone(430 + validation.size * 55, 0.09, 'triangle', 720, 0.11);
      }
      updateValidationUI();
      haptic(validation.has(token) ? [12, 24, 18] : 9);
    });
  });

  $$('.loadout article').forEach((card) => {
    const activate = () => {
      const module = card.dataset.module;
      const target = Number(card.dataset.scrollPixel || 760);
      if (validation.has(module)) {
        showToast(`${module.toUpperCase()} IS ALREADY ARMED.`);
      } else {
        showToast(`RETURNING TO THE ${module.toUpperCase()} PROOF CARD.`);
      }
      scrollPosterTo(target);
      tone(280, 0.06, 'triangle', 390, 0.07);
    };
    card.addEventListener('click', activate);
    card.addEventListener('keydown', (event) => {
      if (event.key === 'Enter' || event.key === ' ') {
        event.preventDefault();
        activate();
      }
    });
  });

  $$('.strain-hotspot').forEach((button) => {
    button.addEventListener('click', () => {
      const name = button.dataset.strain;
      const data = strainData[name];
      if (!dialog || !data) return;
      $('#strainName').textContent = name;
      $('#strainDescription').textContent = data.description;
      $('#strainTags').innerHTML = data.tags.map((tag) => `<span>${tag}</span>`).join('');
      dialog.style.setProperty('--strain-accent', data.accent || '#a7d324');
      if (typeof dialog.showModal === 'function') dialog.showModal();
      else dialog.setAttribute('open', '');
      tone(380, 0.06, 'square', 520, 0.07);
    });
  });

  $('#closeStrain')?.addEventListener('click', () => dialog?.close());
  dialog?.addEventListener('click', (event) => {
    if (event.target === dialog) dialog.close();
  });
  window.addEventListener('keydown', (event) => {
    if (event.key !== 'Escape') return;
    if (dialog?.open) {
      event.preventDefault();
      dialog.close();
      return;
    }
    if (menu && !menu.hidden) {
      event.preventDefault();
      closeMenu();
    }
  });

  menuClose?.addEventListener('click', closeMenu);
  $$('#posterMenu a').forEach((link) => {
    link.addEventListener('click', (event) => {
      closeMenu();
      if (link.dataset.scrollY) {
        event.preventDefault();
        scrollPosterTo(Number(link.dataset.scrollY));
      }
    });
  });

  if (window.matchMedia('(pointer:fine)').matches) {
    const armHover = () => {
      if (document.body.classList.contains('locked')) return;
      document.body.classList.add('hotspot-hover-armed');
      window.removeEventListener('pointermove', armHover);
    };
    window.addEventListener('pointermove', armHover, { passive: true });
  }
}

function initPageProgress() {
  const fill = $('#pageProgressFill');
  if (!fill) return;
  const update = () => {
    const max = document.documentElement.scrollHeight - window.innerHeight;
    const progress = max > 0 ? window.scrollY / max : 0;
    fill.style.height = `${clamp(progress, 0, 1) * 100}%`;
  };
  update();
  window.addEventListener('scroll', update, { passive: true });
}

    if (type === 'flanker') {

      enemy.y = 100 + (lane % 5) * 44;
      enemy.vx = direction * random(95, 140);
      enemy.vy = random(14, 26);
      enemy.width = enemy.height = 20;
      enemy.health = 1;
      enemy.score = 180;
    }
    if (type === 'shield') {
      enemy.width = enemy.height = 31;
      enemy.health = 5;
      enemy.shield = 2;
      enemy.vy = random(12, 25);
      enemy.score = 240;
    }
    if (type === 'disruptor') {
      enemy.width = enemy.height = 26;
      enemy.health = 3;
      enemy.vy = random(17, 30);
      enemy.cooldown = random(0.75, 1.25);
      enemy.score = 280;
    }
    this.enemies.push(enemy);
  },

  spawnBoss() {
    this.boss = {
      type: 'boss', x: this.width / 2, y: 110, vx: 120, vy: 0,
      width: this.width < 520 ? 112 : 150, height: this.width < 520 ? 66 : 82,
      health: this.width < 520 ? 96 : 125, maxHealth: this.width < 520 ? 96 : 125,
      cooldown: 0.75, phase: 0, score: 4000, dead: false, hit: 0
    };
    this.enemies.push(this.boss);
    tone(120, 0.38, 'sawtooth', 70, 0.16);
  },

  switchWeapon(direction) {
    this.currentWeapon = (this.currentWeapon + direction + this.weaponNames.length) % this.weaponNames.length;
    this.updateHud();
    tone(280 + this.currentWeapon * 80, 0.05, 'triangle', 360 + this.currentWeapon * 80, 0.06);
    haptic(7);
  },

  firePlayer() {
    const player = this.player;
    if (player.cooldown > 0) return;
    const locked = validation.size;
    if (this.currentWeapon === 0) {
      this.bullets.push({ x: player.x, y: player.y - 26, vx: 0, vy: -570, radius: 4, damage: locked >= 1 ? 2.2 : 1.6, color: '#c8ff38', kind: 'precision', pierce: 0 });
      player.cooldown = 0.14;
      tone(430, 0.035, 'square', 350, 0.055);
    } else if (this.currentWeapon === 1) {
      [-0.27, 0, 0.27].forEach((angle) => {
        this.bullets.push({ x: player.x, y: player.y - 22, vx: Math.sin(angle) * 230, vy: -Math.cos(angle) * 470, radius: 4, damage: locked >= 2 ? 1.6 : 1.15, color: '#6da5bd', kind: 'spread', pierce: 0 });
      });
      player.cooldown = 0.24;
      tone(520, 0.045, 'triangle', 430, 0.055);
    } else if (this.currentWeapon === 2) {
      this.bullets.push({ x: player.x, y: player.y - 28, vx: 0, vy: -430, radius: 8, damage: locked >= 3 ? 3.7 : 2.5, color: '#b9a4cb', kind: 'root', pierce: 2 });
      player.shield = Math.min(2, player.shield + 0.6);
      player.cooldown = 0.34;
      tone(310, 0.06, 'triangle', 220, 0.06);
    } else {
      this.bullets.push({ x: player.x, y: player.y - 30, vx: 0, vy: -650, radius: 5, damage: locked >= 4 ? 4.6 : 3.1, color: '#f7f2e7', kind: 'ash', pierce: 99 });
      player.cooldown = 0.28;
      noise(0.035, 0.035, 1200);
      tone(670, 0.04, 'sawtooth', 560, 0.05);
    }
  },

  whiteAsh() {
    if (!this.running || this.paused || this.ash < 100) return;
    this.ash = 0;
    this.enemyBullets = [];
    this.enemies.forEach((enemy) => {
      enemy.health -= enemy.type === 'boss' ? 18 : 7;
      enemy.hit = 0.18;
    });
    const particleCount = this.quality === 'high' ? 64 : (this.quality === 'medium' ? 42 : 24);
    for (let index = 0; index < particleCount; index += 1) {
      const angle = index / particleCount * Math.PI * 2;
      this.particles.push({ x: this.player.x, y: this.player.y, vx: Math.cos(angle) * random(90, 310), vy: Math.sin(angle) * random(90, 310), life: random(0.35, 0.85), age: 0, size: random(2, 7), color: '#f1eee5' });
    }
    this.rings.push({ x: this.player.x, y: this.player.y, radius: 24, max: Math.hypot(this.width, this.height), life: 0.75, age: 0, color: '#f3efe4', width: 8 });
    this.shake = this.reduceShake ? 0 : 19;
    noise(0.17, 0.08, 650);
    tone(210, 0.22, 'square', 940, 0.15);
    haptic([20, 20, 35]);
    this.updateHud();
  },

  damagePlayer(amount) {
    if (this.player.shield > 0) {
      amount *= 0.32;
      this.player.shield = Math.max(0, this.player.shield - 0.65);
    }
    this.player.health = clamp(this.player.health - amount, 0, 100);
    this.player.flash = 0.15;
    this.combo = 1;
    this.comboTimer = 0;
    this.shake = this.reduceShake ? 0 : Math.max(this.shake, 11);
    noise(0.05, 0.065, 430);
    tone(180, 0.07, 'sawtooth', 110, 0.09);
    haptic(18);
    this.updateHud();
    if (this.player.health <= 0) this.finish(false);
  },

  addScore(base) {
    this.score += base * this.combo;
    this.combo = clamp(this.combo + 1, 1, 9);
    this.comboTimer = 2.5;
    this.ash = clamp(this.ash + base / 17, 0, 100);
    this.updateHud();
  },

  killEnemy(enemy) {
    if (!enemy || enemy.dead) return;
    enemy.dead = true;
    this.addScore(enemy.score || 100);
    const particleCount = this.quality === 'high' ? 16 : 9;
    for (let index = 0; index < particleCount; index += 1) {
      this.particles.push({ x: enemy.x, y: enemy.y, vx: random(-150, 150), vy: random(-150, 150), life: random(0.24, 0.68), age: 0, size: random(2, 6), color: enemy.type === 'boss' ? '#fff0c0' : (enemy.type === 'shield' ? '#b9a4cb' : '#a7d324') });
    }
    this.rings.push({ x: enemy.x, y: enemy.y, radius: 7, max: enemy.type === 'boss' ? 180 : 58, life: enemy.type === 'boss' ? 0.55 : 0.22, age: 0, color: enemy.type === 'boss' ? '#f3efe4' : '#a7d324', width: enemy.type === 'boss' ? 8 : 3 });
    noise(enemy.type === 'boss' ? 0.12 : 0.028, enemy.type === 'boss' ? 0.09 : 0.03, 800);
    if (enemy.type === 'boss') {
      this.boss = null;
      this.enemies = this.enemies.filter((item) => item !== enemy);
      this.finish(true);
    }
  },

  update(delta) {
    const player = this.player;
    player.cooldown = Math.max(0, player.cooldown - delta);
    player.shield = Math.max(0, player.shield - delta * 0.08);
    player.flash = Math.max(0, player.flash - delta);
    this.shake = Math.max(0, this.shake - delta * 30);
    this.waveBannerTimer = Math.max(0, this.waveBannerTimer - delta);

    let dx = 0;
    let dy = 0;
    if (this.keys.arrowleft || this.keys.a) dx -= 1;
    if (this.keys.arrowright || this.keys.d) dx += 1;
    if (this.keys.arrowup || this.keys.w) dy -= 1;
    if (this.keys.arrowdown || this.keys.s) dy += 1;
    if (this.pointer) {
      const maxStep = player.speed * delta;
      player.x += clamp(this.pointer.x - player.x, -maxStep, maxStep);
      player.y += clamp(this.pointer.y - player.y, -maxStep, maxStep);
    } else if (dx || dy) {
      const magnitude = Math.hypot(dx, dy) || 1;
      player.x += dx / magnitude * player.speed * delta;
      player.y += dy / magnitude * player.speed * delta;
    }
    const maxPlayerY = this.touch ? Math.min(this.height - 160, this.height * 0.78) : this.height - 36;
    player.x = clamp(player.x, player.radius + 5, this.width - player.radius - 5);
    player.y = clamp(player.y, 92, Math.max(100, maxPlayerY));

    if (this.keys[' '] || this.mouseFiring || this.touch) this.firePlayer();
    if (this.comboTimer > 0) this.comboTimer -= delta;
    else this.combo = 1;

    const waveDefinition = this.waves[this.wave - 1];
    if (this.spawnQueue.length) {
      this.waveTimer -= delta;
      if (this.waveTimer <= 0) {
        this.spawnEnemy(this.spawnQueue.shift());
        this.waveTimer = waveDefinition?.delay || 0.65;
      }
    } else if (!this.boss && this.enemies.length === 0 && this.wave < 5) {
      this.queueWave(this.wave + 1);
    }

    this.stars.forEach((star) => {
      star.y += star.speed * delta;
      if (star.y > this.height) {
        star.y = -3;
        star.x = Math.random() * this.width;
      }
    });
    this.bullets.forEach((bullet) => { bullet.x += bullet.vx * delta; bullet.y += bullet.vy * delta; });
    this.enemyBullets.forEach((bullet) => { bullet.x += bullet.vx * delta; bullet.y += bullet.vy * delta; });
    this.bullets = this.bullets.filter((bullet) => bullet.y > -70 && bullet.y < this.height + 70 && bullet.x > -70 && bullet.x < this.width + 70 && !bullet.dead);
    this.enemyBullets = this.enemyBullets.filter((bullet) => bullet.y < this.height + 80 && bullet.y > -80 && bullet.x > -80 && bullet.x < this.width + 80 && !bullet.dead);

    this.enemies.forEach((enemy) => {
      enemy.phase += delta;
      enemy.cooldown -= delta;
      enemy.hit = Math.max(0, enemy.hit - delta);
      if (enemy.type === 'basic') {
        enemy.y += enemy.vy * delta;
        enemy.x += Math.sin(enemy.phase * 2.2 + enemy.lane) * 21 * delta;
      } else if (enemy.type === 'flanker') {
        enemy.x += enemy.vx * delta;
        enemy.y += enemy.vy * delta + Math.sin(enemy.phase * 4.7) * 28 * delta;
        if (enemy.x < -38 || enemy.x > this.width + 38) enemy.vx *= -1;
      } else if (enemy.type === 'shield') {
        enemy.y += enemy.vy * delta;
        enemy.x += Math.sin(enemy.phase * 1.8 + enemy.lane) * 17 * delta;
      } else if (enemy.type === 'disruptor') {
        enemy.y += enemy.vy * delta;
        enemy.x += Math.cos(enemy.phase * 3.1 + enemy.lane) * 50 * delta;
      } else if (enemy.type === 'boss') {
        enemy.x += enemy.vx * delta;
        if (enemy.x < enemy.width * 0.58 || enemy.x > this.width - enemy.width * 0.58) enemy.vx *= -1;
      }

      if (enemy.cooldown <= 0) {
        if (enemy.type === 'basic' || enemy.type === 'shield') {
          this.enemyBullets.push({ x: enemy.x, y: enemy.y + enemy.height * 0.55, vx: 0, vy: random(180, 235), radius: 5, color: '#ef6d4d', kind: 'seed' });
        } else if (enemy.type === 'flanker') {
          this.enemyBullets.push({ x: enemy.x, y: enemy.y + 8, vx: random(-70, 70), vy: random(215, 270), radius: 4, color: '#ef8d43', kind: 'fast' });
        } else if (enemy.type === 'disruptor') {
          [-0.28, 0, 0.28].forEach((angle) => {
            this.enemyBullets.push({ x: enemy.x, y: enemy.y + 8, vx: Math.sin(angle) * 190, vy: Math.cos(angle) * 235, radius: 4, color: '#f15cae', kind: 'percent' });
          });
        } else if (enemy.type === 'boss') {
          for (let index = -2; index <= 2; index += 1) {
            this.enemyBullets.push({ x: enemy.x + index * 16, y: enemy.y + enemy.height * 0.4, vx: index * 42, vy: 240, radius: 6, color: '#efe9dc', kind: 'corporate' });
          }
        }
        enemy.cooldown = enemy.type === 'boss' ? 0.56 : random(0.9, 1.8);
        noise(0.018, 0.012, 1300);
      }

      if (enemy.y > this.height + 70) enemy.dead = true;
      const collisionDistance = Math.hypot(enemy.x - player.x, enemy.y - player.y);
      if (collisionDistance < Math.max(enemy.width, enemy.height) * 0.42 + player.radius) {
        this.damagePlayer(enemy.type === 'boss' ? 20 : 12);
        if (enemy.type !== 'boss') enemy.dead = true;
        else {
          enemy.health -= 1;
          enemy.y = Math.min(enemy.y, 135);
        }
      }
    });

    this.bullets.forEach((bullet) => {
      for (const enemy of this.enemies) {
        if (enemy.dead || bullet.dead) continue;
        if (Math.abs(bullet.x - enemy.x) < enemy.width * 0.55 + bullet.radius && Math.abs(bullet.y - enemy.y) < enemy.height * 0.55 + bullet.radius) {
          let damage = bullet.damage;
          if (enemy.shield > 0) {
            enemy.shield -= 1;
            damage *= 0.55;
          }
          enemy.health -= damage;
          enemy.hit = 0.1;
          if (bullet.pierce > 0) bullet.pierce -= 1;
          else bullet.dead = true;
          const impactCount = this.quality === 'high' ? 6 : 3;
          for (let index = 0; index < impactCount; index += 1) this.particles.push({ x: bullet.x, y: bullet.y, vx: random(-100, 100), vy: random(-100, 100), life: 0.24, age: 0, size: random(1, 4), color: bullet.color });
          this.rings.push({ x: bullet.x, y: bullet.y, radius: 3, max: 24, life: 0.14, age: 0, color: bullet.color, width: 2 });
          if (enemy.health <= 0) this.killEnemy(enemy);
        }
      }
    });

    this.enemyBullets.forEach((bullet) => {
      if (!bullet.dead && Math.hypot(bullet.x - player.x, bullet.y - player.y) < player.radius + bullet.radius) {
        bullet.dead = true;
        this.damagePlayer(bullet.radius > 5 ? 16 : 9);
      }
    });

    this.enemies = this.enemies.filter((enemy) => !enemy.dead && enemy.health > 0);
    this.particles.forEach((particle) => {
      particle.age += delta;
      particle.x += particle.vx * delta;
      particle.y += particle.vy * delta;
      particle.vx *= 0.985;
      particle.vy *= 0.985;
    });
    this.particles = this.particles.filter((particle) => particle.age < particle.life);
    this.rings.forEach((ring) => {
      ring.age += delta;
      ring.radius += (ring.max - ring.radius) * Math.min(1, delta * 7.5);
    });
    this.rings = this.rings.filter((ring) => ring.age < ring.life);
    this.updateHud();
  },

  updateHud() {
    const score = $('#scoreReadout');
    const combo = $('#comboReadout');
    const wave = $('#waveReadout');
    const health = $('#healthBar');
    const ash = $('#ashBar');
    const weapon = $('#weaponReadout');
    if (score) score.textContent = String(Math.floor(this.score)).padStart(6, '0');
    if (combo) combo.textContent = `x${this.combo}`;
    if (wave) wave.textContent = `${Math.min(this.wave, 5)}/5`;
    if (health) health.style.width = `${clamp(this.player.health, 0, 100)}%`;
    if (ash) ash.style.width = `${clamp(this.ash, 0, 100)}%`;
    if (weapon) weapon.textContent = this.weaponNames[this.currentWeapon];
  },

  draw() {
    const context = this.context;
    if (!context) return;
    const shakeX = this.reduceShake ? 0 : random(-this.shake, this.shake);
    const shakeY = this.reduceShake ? 0 : random(-this.shake, this.shake);
    context.save();
    context.setTransform(this.dpr, 0, 0, this.dpr, 0, 0);
    context.clearRect(0, 0, this.width, this.height);
    context.translate(shakeX, shakeY);

    const gradient = context.createLinearGradient(0, 0, 0, this.height);
    gradient.addColorStop(0, '#020402');
    gradient.addColorStop(0.62, '#081007');
    gradient.addColorStop(1, '#020302');
    context.fillStyle = gradient;
    context.fillRect(0, 0, this.width, this.height);

    if (this.posterImage?.complete) {
      const sourceWidth = this.posterImage.naturalWidth || 864;
      const sourceHeight = this.posterImage.naturalHeight || 1536;
      context.save();
      context.globalAlpha = 0.13;
      context.filter = 'saturate(.72) contrast(1.12) brightness(.64)';
      context.drawImage(this.posterImage, 300, 0, 564, 730, 0, 0, this.width, this.height * 0.7);
      context.globalAlpha = 0.32;
      context.filter = 'saturate(.72) brightness(.58)';
      context.drawImage(this.posterImage, 0, 475, sourceWidth, 420, 0, this.height * 0.67, this.width, this.height * 0.33);
      context.restore();
    }

    this.stars.forEach((star) => {
      context.globalAlpha = star.alpha;
      context.fillStyle = '#e6efcf';
      context.fillRect(star.x, star.y, star.size, star.size * 1.5);
    });
    context.globalAlpha = 1;

    context.strokeStyle = 'rgba(199,249,61,.13)';
    context.lineWidth = 1;
    for (let line = 0; line < 7; line += 1) {
      context.beginPath();
      context.moveTo(0, this.height - 80 - line * 24);
      for (let x = 0; x <= this.width; x += 28) {
        context.lineTo(x, this.height - 75 - line * 24 + Math.sin(x / 85 + performance.now() / 1500 + line) * 5);
      }
      context.stroke();
    }

    context.save();
    context.globalAlpha = 0.35;
    context.strokeStyle = '#d8d0bf';
    context.lineWidth = 4;
    context.beginPath();
    context.moveTo(this.width * 0.14, 120);
    context.quadraticCurveTo(this.width * 0.08, this.height * 0.46, this.width * 0.16, this.height * 0.78);
    context.stroke();
    context.beginPath();
    context.moveTo(this.width * 0.86, 120);
    context.quadraticCurveTo(this.width * 0.92, this.height * 0.46, this.width * 0.84, this.height * 0.78);
    context.stroke();
    context.restore();

    this.drawPlayer(context);
    this.bullets.forEach((bullet) => this.drawProjectile(context, bullet));
    this.enemyBullets.forEach((bullet) => this.drawEnemyProjectile(context, bullet));
    this.enemies.forEach((enemy) => this.drawEnemy(context, enemy));
    this.particles.forEach((particle) => {
      const alpha = 1 - particle.age / particle.life;
      context.globalAlpha = alpha;
      context.fillStyle = particle.color;
      context.fillRect(particle.x - particle.size / 2, particle.y - particle.size / 2, particle.size, particle.size);
    });
    context.globalAlpha = 1;
    this.rings.forEach((ring) => {
      const alpha = 1 - ring.age / ring.life;
      context.globalAlpha = alpha;
      context.strokeStyle = ring.color;
      context.lineWidth = ring.width;
      context.beginPath();
      context.arc(ring.x, ring.y, ring.radius, 0, Math.PI * 2);
      context.stroke();
    });
    context.globalAlpha = 1;

    if (this.waveBannerTimer > 0) {
      const alpha = clamp(this.waveBannerTimer, 0, 1);
      context.save();
      context.globalAlpha = alpha;
      context.translate(this.width / 2, this.height * 0.22);
      context.rotate(-0.015);
      const textWidth = Math.min(this.width - 40, 520);
      context.fillStyle = '#e6ddcc';
      context.fillRect(-textWidth / 2, -28, textWidth, 58);
      context.fillStyle = '#11130f';
      context.textAlign = 'center';
      context.font = `900 ${this.width < 520 ? 20 : 26}px Impact, sans-serif`;
      context.fillText(this.waveBanner, 0, 8);
      context.restore();
    }

    if (!this.reduceFlash && this.player.flash > 0) {
      context.globalAlpha = this.player.flash * 2;
      context.fillStyle = '#ef4b3d';
      context.fillRect(0, 0, this.width, this.height);
      context.globalAlpha = 1;
    }
    context.restore();
  },

  drawPlayer(context) {
    const player = this.player;
    context.save();
    context.translate(player.x, player.y);
    if (player.shield > 0) {
      context.strokeStyle = `rgba(185,164,203,${0.4 + player.shield * 0.2})`;
      context.lineWidth = 4;
      context.beginPath();
      context.arc(0, 0, player.radius + 13, 0, Math.PI * 2);
      context.stroke();
    }
    context.fillStyle = '#4d6b35';
    context.beginPath();
    context.moveTo(0, -player.radius * 1.15);
    context.lineTo(player.radius * 0.9, player.radius * 0.8);
    context.lineTo(0, player.radius * 0.35);
    context.lineTo(-player.radius * 0.9, player.radius * 0.8);
    context.closePath();
    context.fill();
    context.fillStyle = '#8f7899';
    for (let index = 0; index < 9; index += 1) {
      const angle = index / 9 * Math.PI * 2;
      context.beginPath();
      context.moveTo(Math.cos(angle) * 3, Math.sin(angle) * 3);
      context.lineTo(Math.cos(angle - 0.2) * player.radius * 0.7, Math.sin(angle - 0.2) * player.radius * 0.7);
      context.lineTo(Math.cos(angle + 0.2) * player.radius * 0.7, Math.sin(angle + 0.2) * player.radius * 0.7);
      context.closePath();
      context.fill();
    }
    context.fillStyle = '#a7d324';
    context.beginPath();
    context.arc(0, 0, player.radius * 0.34, 0, Math.PI * 2);
    context.fill();
    context.fillStyle = '#e77b39';
    for (let index = 0; index < 6; index += 1) {
      context.fillRect(random(-player.radius * 0.38, player.radius * 0.38), random(-player.radius * 0.38, player.radius * 0.38), 2, 7);
    }
    context.restore();
  },

  drawProjectile(context, bullet) {
    context.save();
    context.translate(bullet.x, bullet.y);
    context.fillStyle = bullet.color;
    if (bullet.kind === 'precision') {
      context.fillRect(-2, -12, 4, 24);
      context.globalAlpha = 0.25;
      context.fillRect(-5, 4, 10, 18);
    } else if (bullet.kind === 'spread') {
      context.beginPath();
      context.moveTo(0, -8);
      context.lineTo(5, 6);
      context.lineTo(-5, 6);
      context.closePath();
      context.fill();
    } else if (bullet.kind === 'root') {
      context.strokeStyle = bullet.color;
      context.lineWidth = 3;
      context.beginPath();
      context.moveTo(0, -8);
      context.lineTo(-6, 0);
      context.lineTo(0, 8);
      context.lineTo(6, 0);
      context.closePath();
      context.stroke();
    } else {
      context.beginPath();
      context.arc(0, 0, bullet.radius, 0, Math.PI * 2);
      context.fill();
      context.globalAlpha = 0.25;
      context.beginPath();
      context.arc(0, 0, bullet.radius * 2.1, 0, Math.PI * 2);
      context.fill();
    }
    context.restore();
  },

  drawEnemyProjectile(context, bullet) {
    context.save();
    context.translate(bullet.x, bullet.y);
    context.fillStyle = bullet.color;
    if (bullet.kind === 'percent') {
      context.font = '900 13px Arial';
      context.textAlign = 'center';
      context.fillText('%', 0, 5);
    } else if (bullet.kind === 'corporate') {
      context.rotate(Math.PI / 4);
      context.fillRect(-bullet.radius * 0.7, -bullet.radius * 0.7, bullet.radius * 1.4, bullet.radius * 1.4);
    } else if (bullet.kind === 'seed') {
      context.beginPath();
      context.ellipse(0, 0, bullet.radius * 0.7, bullet.radius * 1.2, 0, 0, Math.PI * 2);
      context.fill();
    } else {
      context.beginPath();
      context.arc(0, 0, bullet.radius, 0, Math.PI * 2);
      context.fill();
    }
    context.restore();
  },

  drawEnemy(context, enemy) {
    context.save();
    context.translate(enemy.x, enemy.y);
    if (enemy.hit > 0 && !this.reduceFlash) context.globalAlpha = 0.45;
    if (enemy.type === 'basic') {
      context.fillStyle = '#8a8d85';
      context.fillRect(-enemy.width / 2, -enemy.height / 2, enemy.width, enemy.height);
      context.fillStyle = '#353832';
      context.fillRect(-enemy.width * 0.28, -enemy.height * 0.19, enemy.width * 0.56, enemy.height * 0.38);
      context.fillStyle = '#ef4b3d';
      context.fillRect(enemy.width * 0.24, enemy.height * 0.25, 4, 4);
    } else if (enemy.type === 'flanker') {
      context.fillStyle = '#bd6537';
      context.beginPath();
      context.moveTo(0, -enemy.height * 0.6);
      context.lineTo(enemy.width * 0.58, enemy.height * 0.5);
      context.lineTo(0, enemy.height * 0.25);
      context.lineTo(-enemy.width * 0.58, enemy.height * 0.5);
      context.closePath();
      context.fill();
      context.fillStyle = '#e6ddcc';
      context.beginPath();
      context.moveTo(0, -enemy.height * 0.35);
      context.lineTo(enemy.width * 0.25, enemy.height * 0.28);
      context.lineTo(-enemy.width * 0.25, enemy.height * 0.28);
      context.closePath();
      context.fill();
    } else if (enemy.type === 'shield') {
      context.fillStyle = '#776182';
      context.fillRect(-enemy.width / 2, -enemy.height / 2, enemy.width, enemy.height);
      context.fillStyle = '#d6ccdf';
      context.fillRect(-enemy.width * 0.27, -enemy.height * 0.28, enemy.width * 0.54, enemy.height * 0.56);
      context.strokeStyle = '#d6ccdf';
      context.lineWidth = 3;
      context.beginPath();
      context.arc(0, 0, enemy.width * 0.68, 0, Math.PI * 2);
      context.stroke();
      context.fillStyle = '#67843e';
      context.font = '900 13px Arial';
      context.textAlign = 'center';
      context.fillText('✓', 0, 5);
    } else if (enemy.type === 'disruptor') {
      context.fillStyle = '#bd3e80';
      context.beginPath();
      for (let point = 0; point < 14; point += 1) {
        const angle = point / 14 * Math.PI * 2;
        const radius = point % 2 === 0 ? enemy.width * 0.62 : enemy.width * 0.34;
        const x = Math.cos(angle) * radius;
        const y = Math.sin(angle) * radius;
        if (point === 0) context.moveTo(x, y); else context.lineTo(x, y);
      }
      context.closePath();
      context.fill();
      context.fillStyle = '#151713';
      context.font = '900 12px Arial';
      context.textAlign = 'center';
      context.fillText('%', 0, 4);
    } else {
      context.fillStyle = '#a6a7a1';
      context.fillRect(-enemy.width / 2, -enemy.height / 2, enemy.width, enemy.height);
      context.fillStyle = '#50524d';
      context.fillRect(-enemy.width * 0.32, -enemy.height * 0.27, enemy.width * 0.64, enemy.height * 0.54);
      context.fillStyle = '#181a17';
      context.font = `900 ${Math.max(10, enemy.height * 0.18)}px Arial`;
      context.textAlign = 'center';
      context.fillText('CROP DIRECTOR', 0, 4);
      context.fillStyle = '#e6ddcc';
      context.fillRect(-enemy.width * 0.42, -enemy.height * 0.73, enemy.width * 0.84, 8);
      context.fillStyle = '#ef4b3d';
      context.fillRect(-enemy.width * 0.42, -enemy.height * 0.73, enemy.width * 0.84 * (enemy.health / enemy.maxHealth), 8);
    }
    context.globalAlpha = 1;
    context.restore();
  },

  loop(time) {
    if (!this.running) return;
    const delta = Math.min(0.033, (time - this.lastTime) / 1000 || 0.016);
    this.lastTime = time;
    if (!this.paused) {
      this.update(delta);
      this.draw();
    }
    this.raf = requestAnimationFrame((nextTime) => this.loop(nextTime));
  },

  togglePause() {
    if (!this.running) return;
    if (this.paused) {
      this.paused = false;
      $('#gameOverlay').hidden = true;
      this.lastTime = performance.now();
    } else {
      this.pause('PAUSED', 'THE HOUSE IS HOLDING.');
    }
  },

  pause(title, text) {
    this.paused = true;
    this.overlay(title, text, [
      { label: 'RESUME', action: 'resume' },
      { label: 'RETURN TO POSTER', action: 'poster' }
    ]);
  },

  finish(won) {
    if (this.ended) return;
    this.ended = true;
    this.running = false;
    cancelAnimationFrame(this.raf);
    const best = Math.max(Number(safeStorage.get('powrhouse-best', '0') || 0), Math.floor(this.score));
    safeStorage.set('powrhouse-best', String(best));
    const bestReadout = $('#bestScore');
    if (bestReadout) bestReadout.textContent = String(best).padStart(6, '0');
    if (won) {
      tone(320, 0.18, 'triangle', 840, 0.16);
      this.overlay('HOUSE SECURED', 'THE GARDEN SURVIVED THE CORPORATE CROP INVASION.', [
        { label: 'RUN IT BACK', action: 'restart' },
        { label: 'VIEW ENDING', action: 'ending' }
      ]);
    } else {
      this.overlay('QUALITY BREACH', 'THE GRAY CROP GOT THROUGH. REVALIDATE AND TRY AGAIN.', [
        { label: 'RESTART', action: 'restart' },
        { label: 'VIEW ENDING', action: 'ending' }
      ]);
    }
  },

  overlay(title, text, buttons) {
    const overlay = $('#gameOverlay');
    if (!overlay) return;
    overlay.hidden = false;
    overlay.innerHTML = `<div class="game-card" role="dialog" aria-modal="true"><h3>${title}</h3><p>${text}</p>${buttons.map((button, index) => `<button class="paper-button ${index === 0 ? 'lime' : 'dark'}" type="button" data-game-action="${button.action}">${button.label}</button>`).join('')}</div>`;
    const overlayButtons = $$('[data-game-action]', overlay);
    overlay.onkeydown = (event) => {
      if (event.key !== 'Tab' || !overlayButtons.length) return;
      const first = overlayButtons[0];
      const last = overlayButtons[overlayButtons.length - 1];
      if (event.shiftKey && document.activeElement === first) { event.preventDefault(); last.focus(); }
      else if (!event.shiftKey && document.activeElement === last) { event.preventDefault(); first.focus(); }
    };
    requestAnimationFrame(() => overlayButtons[0]?.focus());
    overlayButtons.forEach((button) => {
      button.addEventListener('click', () => {
        const action = button.dataset.gameAction;
        if (action === 'resume') {
          this.paused = false;
          overlay.hidden = true;
          this.lastTime = performance.now();
        }
        if (action === 'restart') this.start();
        if (action === 'ending') showEnding(true);
        if (action === 'poster') {
          this.running = false;
          this.paused = false;
          cancelAnimationFrame(this.raf);
          overlay.hidden = true;
          this.shell.hidden = true;
          document.body.classList.remove('game-active');
          $('#arcadeIntro').hidden = false;
          $('#poster').scrollIntoView({ behavior: 'smooth', block: 'start' });
        }
      });
    });
  }
};

function showEnding(scroll = false) {
  Game.running = false;
  cancelAnimationFrame(Game.raf);
  if (Game.shell) Game.shell.hidden = true;
  document.body.classList.remove('game-active');
  const intro = $('#arcadeIntro');
  const ending = $('#ending');
  if (intro) intro.hidden = true;
  if (ending) {
    ending.hidden = false;
    const best = Number(safeStorage.get('powrhouse-best', '0') || 0);
    const bestReadout = $('#bestScore');
    if (bestReadout) bestReadout.textContent = String(best).padStart(6, '0');
    if (scroll) ending.scrollIntoView({ behavior: 'smooth', block: 'start' });
  }
}

function initEnding() {
  $('#restartExperience')?.addEventListener('click', () => {
    document.body.classList.remove('game-active');
    const ending = $('#ending');
    const intro = $('#arcadeIntro');
    if (ending) ending.hidden = true;
    if (intro) intro.hidden = false;
    $('#poster').scrollIntoView({ behavior: 'smooth', block: 'start' });
  });
}

async function hydratePosterFromVault() {
  const poster = $('#posterSource');
  const vault = $('#posterVault');
  if (!poster || !vault) return;
  const started = performance.now();
  const findPoster = () => {
    try {
      const doc = vault.contentDocument;
      if (!doc) return '';
      const preferred = doc.querySelector('#posterSource, #posterImage, img.poster-image');
      if (preferred?.src?.startsWith('data:image/')) return preferred.src;
      const candidates = [...doc.images];
      const exact = candidates.find((img) => img.naturalWidth === 864 && img.naturalHeight === 1536 && img.src.startsWith('data:image/'));
      if (exact) return exact.src;
      const sourceHtml = doc.documentElement?.outerHTML || '';
      const match = sourceHtml.match(/data:image\/(?:jpeg|jpg|webp|png);base64,[A-Za-z0-9+/=]+/);
      return match?.[0] || '';
    } catch (error) {
      return '';
    }
  };
  let source = findPoster();
  while (!source && performance.now() - started < 30000) {
    await new Promise((resolve) => window.setTimeout(resolve, 120));
    source = findPoster();
  }
  if (!source) throw new Error('The exact poster artwork could not be recovered from the protected legacy payload.');
  poster.src = source;
  await poster.decode().catch(() => {});
  $$('[data-poster-copy]').forEach((image) => { image.src = source; });
  vault.remove();
}

async function init() {
  await hydratePosterFromVault();
  const posterSource = $('#posterSource');
  if (posterSource) {
    $$('[data-poster-copy]').forEach((image) => { image.src = posterSource.src; });
  }
  initAgeGate();
  initPosterInteractions();
  initPageProgress();
  updateValidationUI();
  Game.init();
  initEnding();
  const bestReadout = $('#bestScore');
  if (bestReadout) bestReadout.textContent = String(Number(safeStorage.get('powrhouse-best', '0') || 0)).padStart(6, '0');
}

init().catch((error) => {
  document.body.classList.add('boot-error');
  document.body.style.visibility = 'visible';
  showToast(error.message, 8000);
});
