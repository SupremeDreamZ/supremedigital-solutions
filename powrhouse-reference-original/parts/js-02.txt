
            this.enemyBullets.push({ x: enemy.x, y: enemy.y, vx: Math.sin(angle) * 190, vy: Math.cos(angle) * 235, radius: 4, color: '#f15cae', kind: 'percent' });
          });
        } else if (enemy.type === 'boss') {
          for (let index = -2; index <= 2; index += 1) {
            this.enemyBullets.push({ x: enemy.x + index * 16, y: enemy.y + 30, vx: index * 44, vy: 245, radius: 6, color: '#fff0e0', kind: 'invoice' });
          }
        }
        enemy.fire = enemy.type === 'boss' ? 0.58 : random(0.82, 1.75);
      }

      if (enemy.y > this.height + 70) {
        enemy.dead = true;
        this.damagePlayer(8);
      }
      if (Math.hypot(enemy.x - player.x, enemy.y - player.y) < enemy.width * 0.48 + player.radius) {
        if (enemy.type === 'boss') {
          this.damagePlayer(24);
          enemy.vx *= -1;
          enemy.y = Math.max(80, enemy.y - 24);
        } else {
          enemy.dead = true;
          this.damagePlayer(12);
        }
      }
    }

    for (const bullet of this.bullets) {
      for (const enemy of this.enemies) {
        if (enemy.dead) continue;
        const hit = Math.abs(bullet.x - enemy.x) < enemy.width * 0.55 + bullet.radius && Math.abs(bullet.y - enemy.y) < enemy.height * 0.55 + bullet.radius;
        if (!hit) continue;
        let damage = bullet.damage;
        if (enemy.shield > 0) {
          enemy.shield -= 1;
          damage *= 0.45;
        }
        enemy.health -= damage;
        enemy.hitFlash = 0.09;
        bullet.pierce -= 1;
        if (bullet.pierce < 0) bullet.dead = true;
        if (enemy.health <= 0) {
          enemy.dead = true;
          this.killEnemy(enemy);
        }
        break;
      }
    }

    for (const bullet of this.enemyBullets) {
      if (Math.hypot(bullet.x - player.x, bullet.y - player.y) < player.radius + bullet.radius) {
        bullet.dead = true;
        this.damagePlayer(bullet.radius >= 6 ? 15 : 9);
      }
    }

    this.bullets = this.bullets.filter((bullet) => !bullet.dead && bullet.y > -60 && bullet.y < this.height + 60 && bullet.x > -60 && bullet.x < this.width + 60);
    this.enemyBullets = this.enemyBullets.filter((bullet) => !bullet.dead && bullet.y < this.height + 80 && bullet.x > -80 && bullet.x < this.width + 80);
    this.enemies = this.enemies.filter((enemy) => !enemy.dead && enemy.health > 0);

    this.particles.forEach((particle) => {
      particle.age += dt;
      particle.x += particle.vx * dt;
      particle.y += particle.vy * dt;
      particle.vx *= 0.985;
      particle.vy *= 0.985;
    });
    this.particles = this.particles.filter((particle) => particle.age < particle.life);
    this.rings.forEach((ring) => { ring.age += dt; });
    this.rings = this.rings.filter((ring) => ring.age < ring.life);
    this.updateHud();
  },

  drawPlayerBullet(context, bullet) {
    context.save();
    context.translate(bullet.x, bullet.y);
    context.fillStyle = bullet.color;
    context.strokeStyle = bullet.color;
    context.lineWidth = Math.max(1.2, bullet.radius * 0.45);
    if (bullet.kind === 'precision') {
      context.fillRect(-bullet.radius * 0.55, -bullet.radius * 2.2, bullet.radius * 1.1, bullet.radius * 4.4);
      context.globalAlpha = 0.22;
      context.fillRect(-bullet.radius * 0.35, 0, bullet.radius * 0.7, 22);
    } else if (bullet.kind === 'spread') {
      context.rotate(Math.atan2(bullet.vy, bullet.vx) + Math.PI / 2);
      context.beginPath();
      context.moveTo(0, -bullet.radius * 1.8);
      context.lineTo(bullet.radius, bullet.radius * 1.1);
      context.lineTo(0, bullet.radius * 0.55);
      context.lineTo(-bullet.radius, bullet.radius * 1.1);
      context.closePath();
      context.fill();
    } else if (bullet.kind === 'root') {
      context.beginPath();
      context.arc(0, 0, bullet.radius, 0, Math.PI * 2);
      context.stroke();
      context.beginPath();
      context.moveTo(0, bullet.radius);
      context.lineTo(-bullet.radius * 0.8, bullet.radius * 2);
      context.moveTo(0, bullet.radius);
      context.lineTo(bullet.radius * 0.8, bullet.radius * 2);
      context.stroke();
    } else {
      context.beginPath();
      for (let index = 0; index < 10; index += 1) {
        const angle = index / 10 * Math.PI * 2;
        const radius = index % 2 ? bullet.radius * 0.8 : bullet.radius * 1.5;
        const x = Math.cos(angle) * radius;
        const y = Math.sin(angle) * radius;
        if (!index) context.moveTo(x, y); else context.lineTo(x, y);
      }
      context.closePath();
      context.fill();
    }
    context.restore();
  },
